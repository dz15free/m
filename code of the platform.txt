async function registerUser() {
    const name = document.getElementById('registerName').value.trim();
    const email = document.getElementById('registerEmail').value.trim();
    const password = document.getElementById('registerPassword').value;

    if (!name || !email || !password) {
        return showGeneralNotification('error', 'الرجاء تعبئة جميع الحقول.');
    }
    if (password.length < 6) {
        return showGeneralNotification('error', 'كلمة المرور يجب أن تكون 6 أحرف على الأقل.');
    }

    try {
        const userCredential = await createUserWithEmailAndPassword(auth, email, password);
        const user = userCredential.user;
        
        // تحديث اسم المستخدم
        await updateProfile(user, { displayName: name });
        
        // *** الخطوة الأهم: إرسال بريد التفعيل ***
        await sendEmailVerification(user);
        
        // تسجيل الخروج لمنع الدخول قبل التفعيل
        await signOut(auth);
        
        // عرض رسالة واضحة للمستخدم
        showGeneralNotification('success', 'تم إنشاء الحساب! تحقق من بريدك لتفعيله ثم عد لتسجيل الدخول.');
        
        // بعد 4 ثوانٍ، قم بتبديل النموذج إلى تسجيل الدخول لتسهيل الأمر
        setTimeout(() => {
            if (typeof showLoginForm === 'function') showLoginForm();
        }, 4000);

    } catch (error) {
        const errorMessages = {
            'auth/email-already-in-use': 'هذا البريد الإلكتروني مسجل بالفعل.',
            'auth/invalid-email': 'صيغة البريد الإلكتروني غير صحيحة.'
        };
        showGeneralNotification('error', errorMessages[error.code] || 'حدث خطأ أثناء إنشاء الحساب.');
        console.error("Registration Error:", error);
    }
}

/**
 * دالة لحذف رسالة معينة من قاعدة البيانات.
 * @param {string} messageId - هوية الرسالة المراد حذفها.
 */
function deleteMessage(messageId) {
    // تحقق من الصلاحية أولاً
    if (!hasPermission('DELETE_MESSAGES')) {
        return showGeneralNotification('error', 'ليس لديك صلاحية حذف الرسائل.');
    }
    if (!window.currentRoom || !messageId) return;

    const messageRef = window.ref(window.database, `rooms/${window.currentRoom}/messages/${messageId}`);
    
    window.remove(messageRef)
        .then(() => {
            showGeneralNotification('info', 'تم حذف الرسالة بنجاح.');
        })
        .catch((error) => {
            console.error("Delete Message Error:", error);
            showGeneralNotification('error', 'فشل حذف الرسالة.');
        });
}

/**
 * دالة لترقية مستخدم إلى مدير أو إزالة صلاحياته.
 * @param {string} userId - هوية المستخدم المراد تغيير دوره.
 */
function toggleAdmin(userId) {
    // تحقق من الصلاحية أولاً
    if (!hasPermission('MANAGE_USERS')) {
        return showGeneralNotification('error', 'ليس لديك صلاحية إدارة المستخدمين.');
    }
    if (!window.currentRoom || !userId) return;

    // لا يمكن للمالك أن يزيل نفسه من الإدارة
    if (userId === window.currentRoomInfo.owner) {
        return showGeneralNotification('warning', 'لا يمكن تغيير دور مالك الغرفة.');
    }

    const adminRef = window.ref(window.database, `rooms/${window.currentRoom}/admins/${userId}`);
    
    window.get(adminRef).then(snapshot => {
        if (snapshot.exists()) {
            // إذا كان المستخدم مديرًا، قم بإزالته من قائمة المدراء
            window.remove(adminRef)
                .then(() => showGeneralNotification('info', 'تمت إزالة المستخدم من الإدارة.'))
                .catch(() => showGeneralNotification('error', 'فشل تحديث الدور.'));
        } else {
            // إذا لم يكن مديرًا، قم بإضافته إلى قائمة المدراء
            window.set(adminRef, true)
                .then(() => showGeneralNotification('success', 'تمت ترقية المستخدم إلى مدير.'))
                .catch(() => showGeneralNotification('error', 'فشل تحديث الدور.'));
        }
    });
}

// استبدل دالة kickUser القديمة بهذه النسخة الجديدة بالكامل
function kickUser(userId) {
    // 1. التحقق من الصلاحية
    if (!hasPermission('MANAGE_USERS')) {
        return showGeneralNotification('error', 'ليس لديك صلاحية إدارة المستخدمين.');
    }
    if (!window.currentRoom || !userId) return;

    // 2. لا يمكن طرد المالك
    if (userId === window.currentRoomInfo.owner) {
        return showGeneralNotification('warning', 'لا يمكن طرد مالك الغرفة.');
    }

    // 3. إنشاء مرجع لقائمة المحظورين
    const bannedUserRef = window.ref(window.database, `rooms/${window.currentRoom}/bannedUsers/${userId}`);
    
    // 4. إضافة المستخدم إلى قائمة المحظورين
    window.set(bannedUserRef, true) // نضع القيمة true لتأكيد الحظر
        .then(() => {
            // 5. بعد التأكد من حظره، قم بإزالته من قائمة الحضور الحالية
            const userRef = window.ref(window.database, `rooms/${window.currentRoom}/users/${userId}`);
            window.remove(userRef);
            
            showGeneralNotification('success', 'تم حظر المستخدم وطرده من الغرفة.');
        })
        .catch((error) => {
            console.error("Kick User Error:", error);
            showGeneralNotification('error', 'حدث خطأ أثناء محاولة حظر المستخدم.');
        });
}

        function saveRoomState(roomId, roomName, videoId = null, videoTime = 0) {
            const roomState = {
                roomId: roomId,
                roomName: roomName,
                videoId: videoId,

        async function restoreRoomState() {
            const savedState = localStorage.getItem('currentRoomState');
            if (!savedState) return;

            try {
                const roomState = JSON.parse(savedState);
                const timeDiff = Date.now() - roomState.timestamp;
                
                // Only restore if saved within last 24 hours
                if (timeDiff > 24 * 60 * 60 * 1000) {
                    clearRoomState();
                    return;
                }

                // Check if room still exists
                const roomRef = window.ref(window.database, `rooms/${roomState.roomId}`);
                const snapshot = await window.get(roomRef);
                
                if (snapshot.exists()) {
                    // Restore room
                    window.currentRoom = roomState.roomId;
                    document.getElementById('roomId').value = roomState.roomName;
                    
                    initializeRoom(roomState.roomId);
                    document.getElementById('exitRoomBtn').classList.remove('hidden');
                    
                    // Restore video if exists
                    await syncWithExistingVideo(roomState.roomId);
                    
                    showMessage('info', `تم استعادة الاتصال بغرفة الدراسة: ${roomState.roomName}`);
                } else {
                    clearRoomState();
                    showMessage('warning', 'الغرفة المحفوظة لم تعد موجودة.');
                }
            } catch (error) {
                console.error('Error restoring room state:', error);
                clearRoomState();
            }
        }

  // استبدل دالة joinRoom القديمة بهذه النسخة الجديدة والمؤمَّنة بالكامل
async function joinRoom() {
    const roomName = document.getElementById('roomId').value.trim();
    if (!roomName) {
        return showGeneralNotification('error', 'الرجاء إدخال اسم غرفة الدراسة.');
    }

    const roomId = roomName.replace(/\s+/g, '-').toLowerCase(); 
    const roomRef = window.ref(window.database, `rooms/${roomId}`);
    
    try {
        const snapshot = await window.get(roomRef);
        if (snapshot.exists()) {
            const roomData = snapshot.val();

            // --- جديد: التحقق من قائمة الحظر قبل الانضمام ---
            // 1. تحقق مما إذا كانت هناك قائمة للمحظورين
            if (roomData.bannedUsers) {
                // 2. تحقق مما إذا كانت هوية المستخدم الحالي موجودة في القائمة
                if (roomData.bannedUsers[window.currentUser.uid]) {
                    // 3. إذا كان محظورًا، امنعه من الدخول
                    return showGeneralNotification('error', 'لقد تم حظرك من هذه الغرفة ولا يمكنك الانضمام إليها.');
                }
            }
            // --- نهاية التحقق من الحظر ---

            // إذا لم يكن محظورًا، استمر في عملية الانضمام العادية
            window.currentRoom = roomId;
            initializeRoom(roomId);
            showGeneralNotification('success', `تم الانضمام بنجاح إلى غرفة الدراسة: ${roomName}`);
            document.getElementById('exitRoomBtn').classList.remove('hidden');
            
            saveRoomState(roomId, roomName);
            
            await syncWithExistingVideo(roomId);

        } else {
            showGeneralNotification('error', 'الغرفة غير موجودة. يرجى إنشاء غرفة جديدة أو التأكد من اسم الغرفة.');
        }
    } catch (error) {
        console.error("Join Room Error:", error);
        showGeneralNotification('error', 'حدث خطأ أثناء البحث عن الغرفة. حاول مرة أخرى.');
    }
}

// استبدل دالة createRoom القديمة بهذه النسخة المحسّنة بالكامل
async function createRoom() {

        return showGeneralNotification('error', 'الرجاء إدخال اسم لغرفة الدراسة الجديدة.');
    }

    const roomId = roomName.replace(/\s+/g, '-').toLowerCase(); 
    
    const roomRef = window.ref(window.database, `rooms/${roomId}`);
    try {
        const snapshot = await window.get(roomRef);
        if (snapshot.exists()) {
            return showGeneralNotification('warning', 'غرفة بهذا الاسم موجودة بالفعل. انضم إليها أو اختر اسماً آخر.');

        const newRoomData = {
            name: roomName,
            type: roomType, // سيتم حفظ "public" أو "private" أو "teacher"

        await window.set(roomRef, newRoomData);

        saveRoomState(roomId, roomName);

// استبدل دالة handleVideoStateChange القديمة بهذه النسخة الجديدة
function handleVideoStateChange(state) {
    if (!state || !state.videoId) return;

    // إذا كان الفيديو الذي وصل من قاعدة البيانات مختلفًا عن الفيديو الحالي
    if (state.videoId !== window.currentVideoId) {
        // قم بتحميل الفيديو الجديد، ومرر له حالة التشغيل الصحيحة (isPlaying)
        loadVideoById(state.videoId, state.currentTime, state.isPlaying);
        return; // توقف هنا، لأن دالة التحميل ستهتم بالباقي
    }

    // إذا كان الفيديو هو نفسه، فقط قم بمزامنة حالته
    if (window.youtubePlayer && typeof window.youtubePlayer.getCurrentTime === 'function') {
        window.isVideoSyncing = true;

        const timeSinceUpdate = (Date.now() - (state.timestamp || Date.now())) / 1000;
        const targetTime = state.currentTime + (state.isPlaying ? timeSinceUpdate : 0);

        // مزامنة الوقت إذا كان الفارق كبيرًا
        if (Math.abs(window.youtubePlayer.getCurrentTime() - targetTime) > 2.5) {
            window.youtubePlayer.seekTo(targetTime, true);
        }

        // مزامنة حالة التشغيل/الإيقاف
        if (state.isPlaying && window.youtubePlayer.getPlayerState() !== YT.PlayerState.PLAYING) {
            window.youtubePlayer.playVideo();
        } else if (!state.isPlaying && window.youtubePlayer.getPlayerState() === YT.PlayerState.PAUSED) {
            // لا تفعل شيئًا إذا كان متوقفًا بالفعل
        } else if (!state.isPlaying && window.youtubePlayer.getPlayerState() !== YT.PlayerState.PAUSED) {
            window.youtubePlayer.pauseVideo();
        }

        setTimeout(() => { window.isVideoSyncing = false; }, 1200);
    }
}

      // استبدل دالة displayMessages القديمة بهذه النسخة الجديدة بالكامل
function displayMessages(messages) {
    const chatMessages = document.getElementById('chatMessages');
    if (!chatMessages) return; // حماية إضافية

    const shouldScroll = chatMessages.scrollTop + chatMessages.clientHeight >= chatMessages.scrollHeight - 20;

    chatMessages.innerHTML = ''; 
    
    if (!messages) {
         chatMessages.innerHTML = '<div class="message-item other"><div class="message-author">مساعد الدراسة</div><div class="message-text">مرحباً بك! ابدأ النقاش أو شارك ملفاً.</div><div class="message-time">الآن</div></div>';
         return;
    }

    // نحتاج إلى مفاتيح الرسائل (messageId) لحذفها
    const messageEntries = Object.entries(messages);
    let newMessagesCount = 0;

    messageEntries.forEach(([messageId, message], index) => {
        const messageDiv = document.createElement('div');
        messageDiv.className = `message-item ${message.userId === window.currentUser.uid ? 'own' : 'other'}`;

        const previousMessage = index > 0 ? messageEntries[index - 1][1] : null;
        const isConsecutive = previousMessage && previousMessage.userId === message.userId && (message.timestamp - previousMessage.timestamp) < 60000;
        if (isConsecutive) messageDiv.classList.add('consecutive');

        const time = new Date(message.timestamp).toLocaleTimeString('fr-DZ', {
            hour: '2-digit',
            minute: '2-digit',
            hourCycle: 'h23'
        });
        
        let messageContent = '';
        const rawText = message.text ? message.text.replace(/(https?:\/\/[^\s]+ )/g, '<a href="$1" target="_blank">$1</a>') : '';
        const textContent = message.text ? `<div class="message-text">${rawText}</div>` : '';

        if (message.type === 'image' && message.file) {
            messageContent = `${textContent}<img src="${message.file.content}" alt="${message.file.name}" onclick="showImageModal('${message.file.content}')">`;
        } else if (message.type === 'file' && message.file) {
            const fileIcon = getFileIcon(message.file.type);
            const fileSize = formatFileSize(message.file.size);
            messageContent = `${textContent}
                <div class="file-card">
                    <div class="file-icon">${fileIcon}</div>
                    <div class="file-info">
                        <span class="file-name">${message.file.name}</span>
                        <span class="file-size">${fileSize}</span>
                    </div>
                    <a href="${message.file.content}" download="${message.file.name}" class="download-btn">
                        <svg viewBox="0 0 24 24" fill="currentColor" width="16" height="16"><path d="M19 9h-4V3H9v6H5l7 7 7-7zM5 18v2h14v-2H5z"/></svg>
                        تحميل
                    </a>
                </div>`;
        } else {
            messageContent = textContent;
        }

        // --- جديد: إضافة زر حذف الرسالة للمدراء ---
        let adminControls = '';
        // إذا كان المستخدم الحالي لديه صلاحية حذف الرسائل
        if (hasPermission('DELETE_MESSAGES')) {
            adminControls = `
                <div class="admin-controls">
                    <button class="admin-btn" title="حذف الرسالة" onclick="deleteMessage('${messageId}')">
                       <i class="fa fa-trash-can"></i>
                    </button>
                </div>
            `;
        }

        // تجميع كود HTML النهائي للرسالة
        messageDiv.innerHTML = `
            <div class="message-author">
                ${getRoleBadge(message.userId)}
                <span>${message.userName}</span>
                ${adminControls}
            </div>
            ${messageContent}
            <div class="message-time">${time}</div>
        `;
        
        chatMessages.appendChild(messageDiv);

        if (message.userId !== window.currentUser.uid && index >= window.lastMessageCount) {
            newMessagesCount++;
        }
    });

    if (shouldScroll) chatMessages.scrollTop = chatMessages.scrollHeight;

    if (newMessagesCount > 0) {
        if (!document.hidden) notificationSound.play().catch(e => {});
        if (!window.isChatOpen) {
            window.unreadMessageCount += newMessagesCount;
            updateChatNotificationBadge();
        }
    }
    
    window.lastMessageCount = messageEntries.length;
}


// دالة مساعدة لاختيار أيقونة بناءً على نوع الملف
function getFileIcon(fileType) {
    if (fileType.includes('pdf')) return '📄';
    if (fileType.includes('word') || fileType.includes('document')) return '📝';
    if (fileType.includes('excel') || fileType.includes('spreadsheet')) return '📊';
    if (fileType.includes('presentation') || fileType.includes('powerpoint')) return '🖥️';
    if (fileType.includes('zip') || fileType.includes('archive')) return '📦';
    if (fileType.includes('audio')) return '🎵';
    if (fileType.includes('video')) return '🎬';
    return '📁'; // أيقونة افتراضية
}

// دالة مساعدة لتنسيق حجم الملف
function formatFileSize(bytes, decimals = 2) {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const dm = decimals < 0 ? 0 : decimals;
    const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
}

 // استبدل دالة sendMessage القديمة بهذه النسخة الجديدة بالكامل
function sendMessage(fileData = null) {
    const messageInput = document.getElementById('messageInput');
    const text = messageInput.value.trim();

    // لا ترسل أي شيء إذا لم يكن هناك نص أو ملف
    if (!text && !fileData) return;
    if (!window.currentRoom) return;

    // بناء كائن الرسالة الأساسي
    const messagePayload = {
        userId: window.currentUser.uid,
        userName: window.currentUser.displayName || window.currentUser.email,
        timestamp: Date.now(),
        type: 'text' // النوع الافتراضي هو نص
    };

    // إذا كان هناك نص، أضفه إلى الرسالة
    if (text) {
        if (text.length > 500) {
            return showGeneralNotification('warning', 'الرسالة طويلة جداً.');
        }
        messagePayload.text = text;
    }

    // إذا كان هناك ملف، قم بتعديل نوع الرسالة وإضافة بيانات الملف
    if (fileData) {
        // إذا كان الملف صورة، حدد النوع كـ "image"
        if (fileData.type.startsWith('image/')) {
            messagePayload.type = 'image';
        } else { // وإلا، فهو "file"
            messagePayload.type = 'file';
        }
        messagePayload.file = fileData; // أضف كائن الملف كاملاً
    }

    // إرسال الرسالة النهائية إلى قاعدة البيانات
    const messagesRef = window.ref(window.database, `rooms/${window.currentRoom}/messages`);
    window.push(messagesRef, messagePayload);

    // تنظيف حقل الإدخال والتركيز عليه
    messageInput.value = '';
    messageInput.focus();
}

      // استبدل دالة handleFileSelect القديمة بهذه النسخة المؤمنة
function handleFileSelect(event) {
    // --- جديد: التحقق من صلاحية إرسال الملفات ---
    // سيتم المنع فقط إذا كانت الغرفة من نوع "teacher" والمستخدم ليس هو الأستاذ
    if (window.currentRoom && !hasPermission('SEND_FILES')) {
        return showGeneralNotification('error', 'فقط الأستاذ يمكنه إرسال ملفات في هذه الغرفة.');
    }

    const file = event.target.files[0];
    if (!file) return;

    const maxSizeInBytes = 25 * 1024 * 1024; 
    if (file.size > maxSizeInBytes) {
        showGeneralNotification('error', 'حجم الملف كبير جداً. الحد الأقصى 25 ميجابايت.');
        event.target.value = '';
        return;
    }

    const reader = new FileReader();
    reader.onload = function(e) {
        showGeneralNotification('info', `جاري إرفاق ملف: ${file.name}`);
        
        const fileData = {
            name: file.name,
            size: file.size,
            type: file.type,
            content: e.target.result
        };
        
        // استدعاء دالة الإرسال مع بيانات الملف
        // ملاحظة: دالة sendMessage نفسها لا تحتاج إلى تعديل
        sendMessage(fileData);
    };
    
    reader.onerror = function() {
        showGeneralNotification('error', 'حدث خطأ أثناء قراءة الملف.');
    };

    reader.readAsDataURL(file);
    event.target.value = '';
}

 // استبدل دالة loadVideo القديمة بهذه النسخة المؤمنة
function loadVideo() {
    // --- جديد: التحقق من الصلاحية قبل تنفيذ أي شيء ---
    if (window.currentRoom && !hasPermission('CONTROL_VIDEO')) {
        return showGeneralNotification('error', 'فقط مدراء الغرفة يمكنهم تغيير الفيديو.');

    if (!videoUrl) {
        showGeneralNotification('warning', 'الرجاء إدخال رابط الدرس التعليمي.');
        return;
    }
    
    const videoId = extractYouTubeId(videoUrl);
    if (!videoId) {
        showGeneralNotification('error', 'رابط يوتيوب غير صحيح. تأكد من صحة الرابط.');
        return;
    }
    
    if (!window.currentRoom) {
        loadVideoById(videoId, 0, true);
        showGeneralNotification('info', 'تم تحميل الفيديو. انضم إلى غرفة لمشاركته مع الآخرين.');

    window.set(videoRef, {
        videoId: videoId,
        isPlaying: true,
        currentTime: 0,
        timestamp: window.serverTimestamp(),

    });

    loadVideoById(videoId, 0, true);

        function extractYouTubeId(url) {
            const patterns = [
                /(?:youtube\.com\/watch\?v=|youtu\.be\/|youtube\.com\/embed\/)([^#&?]*)/,
                /youtube\.com\/live\/([^#&?]*)/,
                /youtube\.com\/watch\?.*v=([^#&?]*)/
            ];
            
            for (const pattern of patterns) {
                const match = url.match(pattern);
                if (match && match[1].length === 11) {
                    return match[1];
                }
            }
            return null;
        }

// استبدل دالة onPlayerStateChange القديمة بهذه النسخة الجديدة بالكامل
function onPlayerStateChange(event) {
    // إذا كان المستخدم لا يملك صلاحية التحكم بالفيديو
    if (!hasPermission('CONTROL_VIDEO')) {
        // تحقق من الحالة المسجلة في قاعدة البيانات (الحالة الصحيحة)
        const correctState = window.currentRoomInfo?.videoState;
        if (correctState) {
            // إذا كانت الحالة الصحيحة هي "تشغيل" والمستخدم حاول إيقافه
            if (correctState.isPlaying && event.data === YT.PlayerState.PAUSED) {
                showGeneralNotification('error', 'فقط مدراء الغرفة يمكنهم التحكم بالفيديو.');
                // أعد تشغيل الفيديو فورًا
                event.target.playVideo();
                return; // أوقف تنفيذ باقي الدالة
            }
            // إذا كانت الحالة الصحيحة هي "إيقاف" والمستخدم حاول تشغيله
            if (!correctState.isPlaying && event.data === YT.PlayerState.PLAYING) {
                showGeneralNotification('error', 'فقط مدراء الغرفة يمكنهم التحكم بالفيديو.');
                // أعد إيقاف الفيديو فورًا
                event.target.pauseVideo();
                return; // أوقف تنفيذ باقي الدالة
            }
        }
    }

    // إذا كان المستخدم لديه صلاحية، أو إذا كان التغيير مطابقًا للحالة الصحيحة، استمر كالمعتاد
    if (window.isVideoSyncing || !window.currentRoom) return;

    const state = {
        videoId: window.currentVideoId,
        currentTime: event.target.getCurrentTime(),
        timestamp: window.serverTimestamp(),
        changedBy: window.currentUser.uid,
        isPlaying: (event.data === YT.PlayerState.PLAYING)
    };

    if (event.data === YT.PlayerState.PLAYING || event.data === YT.PlayerState.PAUSED) {
        const videoRef = window.ref(window.database, `rooms/${window.currentRoom}/videoState`);
        window.set(videoRef, state);
    }
}

            currentTime: window.youtubePlayer.getCurrentTime(),
            timestamp: window.serverTimestamp(),
            changedBy: window.currentUser.uid
        });
    }
}

function toggleFullscreen() {
    const workspace = document.getElementById('roomWorkspace');
    if (!workspace) return;

    // التحقق إذا كان المتصفح ليس في وضع ملء الشاشة
    if (!document.fullscreenElement) {
        // اطلب الدخول في وضع ملء الشاشة
        workspace.requestFullscreen().catch(err => {
            showGeneralNotification('error', `لا يمكن تفعيل وضع الدراسة المركز: ${err.message}`);
        });
    } else {
        // اطلب الخروج من وضع ملء الشاشة
        document.exitFullscreen();
    }
}

// استبدل دالة leaveRoom القديمة بهذه النسخة الجديدة بالكامل
function leaveRoom(forceLeave = false) {
    if (!forceLeave) {
        const confirmModal = document.getElementById('exitConfirmModal');
        if (confirmModal) {
            confirmModal.classList.remove('hidden');
            confirmModal.classList.add('active');
        }
        return;
    }

    if (!window.currentUser || !window.currentRoom) {
        return cleanupUIAndState();
    }

    const roomRef = window.ref(window.database, `rooms/${window.currentRoom}`);
    const userRef = window.ref(window.database, `rooms/${window.currentRoom}/users/${window.currentUser.uid}`);
    
    cleanupRoomListeners();

    // --- المنطق الجديد لحذف الغرفة الفارغة ---
    // 1. احصل على بيانات الغرفة الحالية مرة واحدة
    window.get(roomRef).then(snapshot => {
        if (snapshot.exists()) {
            const roomData = snapshot.val();
            const users = roomData.users || {};
            const userCount = Object.keys(users).length;

            // 2. إذا كان هذا المستخدم هو الأخير في الغرفة
            if (userCount === 1 && users[window.currentUser.uid]) {
                // قم بحذف الغرفة بالكامل
                window.remove(roomRef)
                    .then(() => showGeneralNotification('info', 'تم حذف الغرفة لأنها أصبحت فارغة.'))
                    .catch(err => console.error("Error deleting empty room:", err))
                    .finally(() => cleanupUIAndState());
            } else {
                // 3. إذا لم يكن الأخير، فقط قم بإزالة المستخدم
                window.remove(userRef)
                    .catch(err => console.warn("Could not remove user:", err.message))
                    .finally(() => cleanupUIAndState());
            }
        } else {
            // إذا كانت الغرفة غير موجودة بالفعل، فقط نظف الواجهة
            cleanupUIAndState();
        }
    });
}/**
 * تقوم بتنظيف واجهة المستخدم وحالة التطبيق عند مغادرة الغرفة.
 * تعيد الواجهة إلى شاشة الترحيب الرئيسية وتزيل كل البيانات المتعلقة بالغرفة السابقة.
 */
function cleanupUIAndState() {
    // الخطوة 1: إعادة الواجهة إلى شاشة الترحيب الرئيسية.
    updateUIVisibility(false); 

    // الخطوة 2: إيقاف أي بث مباشر نشط وتنظيف اتصالات WebRTC.
    if (typeof stopStream === 'function') {
        stopStream();
    }

    // الخطوة 3: إعادة تعيين متغيرات التهيئة لجميع الأدوات.
    // هذا يضمن أن الأدوات سيتم تهيئتها من جديد عند دخول غرفة أخرى.
    window.whiteboardInitialized = false;
    window.fileViewerInitialized = false;
    window.liveStreamInitialized = false;

    // الخطوة 4: مسح حالة الغرفة المحفوظة من ذاكرة المتصفح.
    clearRoomState();
    
    // الخطوة 5: إعادة تعيين المتغيرات العامة المتعلقة بالغرفة.
    window.currentRoom = null;
    window.currentRoomInfo = null;
    window.currentVideoId = null;
    
    // *** الإضافة الجديدة هنا ***
    // إخفاء وإفراغ لوحة تحكم الأدوات
    const toolControls = document.getElementById('tool-controls');
    if (toolControls) {
        toolControls.classList.add('hidden');
        toolControls.innerHTML = ''; // إفراغ الأزرار
    }
    // الخطوة 6: تدمير مشغل فيديو يوتيوب لتجنب الأخطاء.
    if (window.youtubePlayer) {
        try {
            window.youtubePlayer.destroy();
        } catch (e) {
            console.warn("Error destroying YouTube player:", e);
        }
        window.youtubePlayer = null;
    }
    
    // الخطوة 7: إعادة تعيين عناصر الواجهة إلى حالتها الافتراضية.
    const elementsToReset = {
        'roomId': (el) => el.value = '',
        'videoUrl': (el) => el.value = '',
        'chatMessages': (el) => el.innerHTML = '',
        'userList': (el) => el.innerHTML = '<li>في انتظار الطلاب...</li>',
        'videoPlayer': (el) => {
            el.classList.remove('no-controls');
            el.innerHTML = `
                <div class="video-player-overlay" id="videoPlayerOverlay"></div>
                <div class="video-placeholder">
                    <i class="fa fa-video"></i>
                    <div>ضع رابط الدرس على اليوتيوب لمشاهدة جماعية</div>
                </div>
                <div class="video-sync-controls" id="videoSyncControls">
                    <button class="sync-btn play" onclick="syncPlay()"><i class="fa fa-play"></i>تشغيل</button>
                    <button class="sync-btn pause" onclick="syncPause()"><i class="fa fa-pause"></i>إيقاف</button>
                </div>`;
        }
    };

    Object.keys(elementsToReset).forEach(id => {
        const el = document.getElementById(id);
        if (el) {
            elementsToReset[id](el);
        }
    });

    // الخطوة 8: إخفاء زر مغادرة الغرفة وإغلاق أي نوافذ تأكيد مفتوحة.
    const exitRoomBtn = document.getElementById('exitRoomBtn');
    if (exitRoomBtn) exitRoomBtn.classList.add('hidden');

    if (typeof cancelExit === 'function') cancelExit();
    if (typeof cancelDeleteRoom === 'function') cancelDeleteRoom();
}

function hasPermission(permission) {
    // إذا لم يكن المستخدم مسجلاً دخوله أو لم ينضم لغرفة، فلا توجد لديه أي صلاحيات
    if (!window.currentUser || !window.currentRoomInfo) {
        return false;
    }

    const { uid } = window.currentUser;
    const { type, owner, admins } = window.currentRoomInfo;

    // تحديد أدوار المستخدم الحالي
    const isOwner = uid === owner;
    const isAdmin = admins && admins[uid]; // التحقق مما إذا كان المستخدم موجودًا في قائمة المدراء
    const isTeacher = type === 'teacher' && isOwner; // فقط مالك "غرفة الأستاذ" هو الأستاذ

    // التحقق من الصلاحية المطلوبة
    switch (permission) {
        // صلاحية التحكم بالفيديو (تشغيل، إيقاف، تغيير الرابط)
        case 'CONTROL_VIDEO':
            if (type === 'public') return true; // الجميع يتحكم في الغرف العامة
            return isOwner || isAdmin; // فقط المالك والمدراء في الغرف الخاصة وغرف الأستاذ

        // صلاحية إرسال الملفات
        case 'SEND_FILES':
            if (type === 'teacher') return isTeacher; // فقط الأستاذ يرسل ملفات في غرفة الأستاذ
            return true; // الجميع يرسل ملفات في الغرف العامة والخاصة

        // صلاحية حذف رسائل الآخرين
        case 'DELETE_MESSAGES':
            return isOwner || isAdmin; // فقط المالك والمدراء

        // صلاحية إدارة المستخدمين (طرد، ترقية لمدير)
        case 'MANAGE_USERS':
            return isOwner || isAdmin; // فقط المالك والمدراء

        // في حالة طلب صلاحية غير معروفة، يتم الرفض كإجراء أمان
        default:
            return false;
    }
}

function getRoleBadge(userId) {
    // إذا لم تكن معلومات الغرفة متاحة، لا تعرض أي شارة
    if (!window.currentRoomInfo) return '';

    const { owner, admins, type } = window.currentRoomInfo;

    // التحقق إذا كان المستخدم هو المالك
    if (userId === owner) {
        // إذا كان مالكًا وغرفة أستاذ، فهو "أستاذ"
        if (type === 'teacher') {
            return `<span class="role-badge teacher">أستاذ</span>`;
        }
        // وإلا، فهو "المالك"
        return `<span class="role-badge owner">المالك</span>`;
    }
    
    // التحقق إذا كان المستخدم مديرًا (وليس المالك)
    if (admins && admins[userId]) {
        return `<span class="role-badge admin">مدير</span>`;
    }
    
    // إذا لم يكن أيًا مما سبق، فهو "طالب"
    return `<span class="role-badge student">طالب</span>`;
}

function deleteRoom() {
    if (!window.currentRoom || !window.currentRoomInfo || window.currentUser.uid !== window.currentRoomInfo.owner) {
        return showGeneralNotification('error', 'فقط مالك الغرفة يمكنه حذفها.');
    }

    const roomRef = window.ref(window.database, `rooms/${window.currentRoom}`);
    
    // حذف الغرفة بالكامل من قاعدة البيانات
    window.remove(roomRef)
        .then(() => {
            showGeneralNotification('success', 'تم حذف الغرفة بنجاح.');
            // استدعاء المغادرة الإجبارية لتنظيف الواجهة
            leaveRoom(true);
            // إخفاء نافذة التأكيد
            cancelDeleteRoom();
        })
        .catch((error) => {
            console.error("Delete Room Error:", error);
            showGeneralNotification('error', 'حدث خطأ أثناء حذف الغرفة.');
        });
}

// استبدل الدالة القديمة بهذه النسخة
async function fetchAndDisplayAvailableRooms() {
    const roomsListDiv = document.getElementById('availableRoomsList');
    const refreshBtn = document.getElementById('refreshRoomsBtn');
    if (!roomsListDiv || !refreshBtn) return;

    // --- الخطوة 1: أضف كلاس التحميل عند بدء الدالة ---
    refreshBtn.classList.add('loading');
    roomsListDiv.innerHTML = '<div class="no-rooms-message">جاري البحث عن غرف متاحة...</div>';

    try {
        const roomsRef = window.ref(window.database, 'rooms');
        const snapshot = await window.get(roomsRef);

        if (!snapshot.exists()) {
            roomsListDiv.innerHTML = '<div class="no-rooms-message">لا توجد أي غرف حاليًا. كن أول من ينشئ غرفة!</div>';
            return;
        }

        const allRooms = snapshot.val();
        const availableRooms = [];
        const fiveMinutesAgo = Date.now() - (5 * 60 * 1000);

        for (const roomId in allRooms) {
            const room = allRooms[roomId];
            if ((room.type === 'public' || room.type === 'teacher') && room.users && Object.keys(room.users).length > 0) {
                const lastJoinedTime = Math.max(...Object.values(room.users).map(u => u.joinedAt || 0));
                if (lastJoinedTime > fiveMinutesAgo) {
                    availableRooms.push({ id: roomId, ...room });
                }
            }
        }

        if (availableRooms.length > 0) {
            roomsListDiv.innerHTML = '';
            availableRooms.forEach(room => {
                const roomCard = document.createElement('div');
                roomCard.className = 'room-card';
                const userCount = Object.keys(room.users || {}).length;
                const roomTypeDisplay = room.type === 'teacher' ? 'غرفة أستاذ' : 'غرفة عامة';

                // ملاحظة: لقد قمت بإزالة أيقونات SVG المتبقية من هنا أيضاً
                roomCard.innerHTML = `
                    <div class="room-card-header">${room.name}</div>
                    <div class="room-card-details">
                        <span class="room-card-detail">
                            <i class="fa fa-user-group"></i>
                            ${userCount} ${userCount > 2 ? 'طلاب' : 'طالب'}
                        </span>
                        <span class="room-card-detail">
                            <i class="fa fa-chalkboard"></i>
                            ${roomTypeDisplay}
                        </span>
                    </div>
                    <button class="btn room-card-join-btn" onclick="joinAvailableRoom('${room.name}')">انضمام سريع</button>
                `;
                roomsListDiv.appendChild(roomCard);
            });
        } else {
            roomsListDiv.innerHTML = '<div class="no-rooms-message">لا توجد غرف نشطة حاليًا.</div>';
        }

    } catch (error) {
        console.error("Error fetching available rooms:", error);
        roomsListDiv.innerHTML = '<div class="no-rooms-message">حدث خطأ أثناء جلب الغرف.</div>';
    } finally {
        // --- الخطوة 2: أزل كلاس التحميل بعد انتهاء كل شيء ---
        // نستخدم تأخير بسيط لإعطاء الانطباع بأن العملية استغرقت وقتاً
        setTimeout(() => {
            refreshBtn.classList.remove('loading');
        }, 500);
    }
}

function updateUIVisibility(inRoom) {
    // تحديد جميع العناصر التي نريد التحكم بها
    const welcomeElements = [
        document.querySelector('.instructions'),
        document.querySelector('.room-management'),
        document.querySelector('.available-rooms-container') // <-- هذا هو السطر الذي أضفناه
    ];
    const roomWorkspace = document.getElementById('roomWorkspace');

    if (inRoom) {
        // --- حالة الانضمام إلى غرفة: إخفاء واجهة الترحيب وإظهار مساحة العمل ---
        welcomeElements.forEach(el => {
            if (el) el.style.display = 'none'; // إخفاء عناصر الواجهة الرئيسية
        });
        if (roomWorkspace) roomWorkspace.classList.remove('hidden'); // إظهار مساحة عمل الغرفة
    } else {
        // --- حالة مغادرة الغرفة: إظهار واجهة الترحيب وإخفاء مساحة العمل ---
        welcomeElements.forEach(el => {
            // نستخدم 'block' أو 'grid' حسب نوع العنصر لضمان العرض الصحيح
            if (el) {
                if (el.classList.contains('available-rooms-container')) {
                    el.style.display = 'block'; // أو 'grid' إذا كان يستخدم grid layout
                } else {
                    el.style.display = 'block';
                }
            }
        });
        if (roomWorkspace) roomWorkspace.classList.add('hidden'); // إخفاء مساحة عمل الغرفة
    }
}

       function initializeRoom(roomId) {
        

    // 1. إخفاء واجهات الصفحة الرئيسية وإظهار مساحة العمل
   updateUIVisibility(true); 
    // 2. تنظيف أي مستمعين قدامى
    cleanupRoomListeners();
    
    // 3. مستمع رئيسي لمعلومات الغرفة
    const roomInfoRef = ref(database, `rooms/${roomId}`);
    const roomInfoListener = onValue(roomInfoRef, (snapshot) => {
        if (snapshot.exists()) {
            window.currentRoomInfo = snapshot.val();
            
            // *** الجزء الجديد والمهم: عرض رسالة الترحيب وتحديث الواجهة ***
            renderWelcomeMessage(); // عرض رسالة الترحيب المخصصة
            renderToolControls();   // عرض لوحة التحكم بناءً على الصلاحيات
            updatePlayerUI();       // تحديث واجهة الفيديو (لإظهار/إخفاء التحكم)
            
            // تحديث قوائم المستخدمين والرسائل
            if (window.currentRoomInfo.users) {
                displayUsers(window.currentRoomInfo.users);
            }
            if (window.currentRoomInfo.messages) {
                displayMessages(window.currentRoomInfo.messages);
            }

            // *** تفعيل مستمع الأداة النشطة ***
            const activeToolRef = ref(database, `rooms/${roomId}/state/activeTool`);
            const activeToolListener = onValue(activeToolRef, (snapshot) => {
                const activeToolId = snapshot.val();
                // إذا كانت هناك أداة نشطة، قم بالتبديل إليها
                // إذا لم تكن هناك أداة (القيمة null)، لا تفعل شيئًا (لتبقى رسالة الترحيب ظاهرة)
                if (activeToolId) {
                    handleToolChange(activeToolId);
                } else {
                    // تأكد من إخفاء الأدوات الأخرى وإظهار رسالة الترحيب
                    handleToolChange('room-welcome-message');
                }
            });
            window.videoSyncListeners.push({ ref: activeToolRef, listener: activeToolListener });

        } else {
            showGeneralNotification('error', 'تم حذف هذه الغرفة من قبل المالك.');
            leaveRoom(true); 
        }
    });
    window.videoSyncListeners.push({ ref: roomInfoRef, listener: roomInfoListener });

    // 4. إعداد معلومات حضور المستخدم في الغرفة
    const userRef = ref(database, `rooms/${roomId}/users/${currentUser.uid}`);
    set(userRef, {
        name: currentUser.displayName || currentUser.email,
        joinedAt: serverTimestamp(),
        status: 'active'
    });

    // ... باقي المستمعين (الطرد، الرسائل، المستخدمين، الفيديو) تبقى كما هي ...
    // 5. مستمع الطرد الفوري
    const bannedUsersRef = ref(database, `rooms/${roomId}/bannedUsers/${currentUser.uid}`);
    const bannedListener = onValue(bannedUsersRef, (snapshot) => {
        if (snapshot.exists()) {
            showGeneralNotification('error', 'لقد تم طردك من هذه الغرفة.');
            leaveRoom(true); 
        }
    });
    window.videoSyncListeners.push({ ref: bannedUsersRef, listener: bannedListener });

    // 6. المستمع الخاص بالرسائل
    const messagesRef = ref(database, `rooms/${roomId}/messages`);
    const messagesListener = onValue(messagesRef, (snapshot) => {
        displayMessages(snapshot.val());
    });
    window.videoSyncListeners.push({ ref: messagesRef, listener: messagesListener });

    // 7. المستمع الخاص بقائمة المستخدمين
    const usersRef = ref(database, `rooms/${roomId}/users`);
    const usersListener = onValue(usersRef, (snapshot) => {
        displayUsers(snapshot.val());
    });
    window.videoSyncListeners.push({ ref: usersRef, listener: usersListener });

    // 8. المستمع الخاص بحالة الفيديو
    const videoStateRef = ref(database, `rooms/${roomId}/videoState`);
    const videoStateListener = onValue(videoStateRef, (snapshot) => {
        const state = snapshot.val();
        if (state && state.changedBy !== currentUser.uid) {
            handleVideoStateChange(state);
        }
    });
    window.videoSyncListeners.push({ ref: videoStateRef, listener: videoStateListener });

    // 9. مزامنة الفيديو الحالي (إذا كان موجودًا)
    syncWithExistingVideo(roomId);
}

      function renderToolControls() {
    const controlsContainer = document.getElementById('tool-controls');
    if (!controlsContainer || !window.currentRoomInfo) return;

    controlsContainer.innerHTML = ''; // تنظيف الأزرار القديمة

    // سنحتاج إلى تعديل دالة hasPermission لاحقًا لتكون أكثر دقة
    // الآن، سنعتمد على النسخة الحالية
    const canControl = hasPermission('CONTROL_VIDEO'); // صلاحية مؤقتة
    if (!canControl) {
        controlsContainer.classList.add('hidden');
        return;
    }

    // الأدوات المتاحة
    const tools = [
        { id: 'youtube', icon: 'fa-video', label: 'فيديو يوتيوب' },
        { id: 'whiteboard', icon: 'fa-chalkboard', label: 'سبورة' },
        { id: 'file-viewer', icon: 'fa-file-alt', label: 'عرض ملف' }
    ];

    // أضف أداة البث المباشر فقط للأستاذ
    if (window.currentRoomInfo.type === 'teacher' && window.currentRoomInfo.owner === window.currentUser.uid) {
        tools.push({ id: 'live-stream', icon: 'fa-video', label: 'بث مباشر' });
    }

    // إنشاء الأزرار
    tools.forEach(tool => {
        const button = document.createElement('button');
        button.className = 'control-btn';
        button.style.flex = '1';
        button.innerHTML = `<i class="fa ${tool.icon}"></i> ${tool.label}`;
        button.onclick = () => switchTool(tool.id); 
        controlsContainer.appendChild(button);
    });

    controlsContainer.classList.remove('hidden');
}

/**
 * الدالة المركزية للتبديل بين أدوات مساحة العمل المختلفة.
 * تقوم بإخفاء جميع الأدوات ثم إظهار الأداة المطلوبة فقط.
 * كما أنها تتأكد من تهيئة كل أداة مرة واحدة فقط وتضيف زر ملء الشاشة.
 * @param {string} activeToolId - هوية الأداة المراد عرضها.
 */
function handleToolChange(activeToolId) {
    // إخفاء جميع حاويات الأدوات.
    document.querySelectorAll('.tool-container').forEach(container => {
        container.classList.add('hidden');
    });

    const toolIdToShow = activeToolId || 'room-welcome-message';
    const containerToShow = document.getElementById(toolIdToShow + '-tool-container');

    if (containerToShow) {
        containerToShow.classList.remove('hidden');
        
        // إضافة زر "وضع الدراسة المركز" ديناميكياً
        const controlsPanel = containerToShow.querySelector('.video-controls');
        if (controlsPanel) {
            // أولاً، قم بإزالة أي نسخة قديمة من الزر لضمان عدم التكرار
            const oldBtn = controlsPanel.querySelector('.fullscreen-btn-unified');
            if (oldBtn) {
                oldBtn.remove();
            }

            // الآن، قم بإنشاء وإضافة الزر الجديد
            const fullscreenBtn = document.createElement('button');
            fullscreenBtn.className = 'control-btn fullscreen-btn-unified';
            fullscreenBtn.innerHTML = '<i class="fa fa-expand"></i> وضع الدراسة المركز';
            fullscreenBtn.onclick = toggleFullscreen;
            
            // أضف الزر كآخر عنصر في شريط الأدوات لضمان الترتيب الصحيح في الشبكة
            controlsPanel.appendChild(fullscreenBtn);
        }

        // تهيئة الأداة عند استخدامها لأول مرة
        switch (toolIdToShow) {
            case 'whiteboard':
                if (!window.whiteboardInitialized) {
                    initializeWhiteboard();
                    window.whiteboardInitialized = true;
                }
                break;
            
            case 'file-viewer':
                if (!window.fileViewerInitialized) {
                    initializeFileViewer();
                    window.fileViewerInitialized = true;
                }
                break;

            case 'live-stream':
                if (!window.liveStreamInitialized) {
                    initializeLiveStream();
                    window.liveStreamInitialized = true;
                }
                break;
        }
    } else {
        // كخيار احتياطي، اعرض رسالة الترحيب
        const welcomeContainer = document.getElementById('room-welcome-message-tool-container');
        if (welcomeContainer) {
            welcomeContainer.classList.remove('hidden');
        }
    }
}

function renderWelcomeMessage() {
    const welcomeContainer = document.getElementById('room-welcome-message');
    if (!welcomeContainer || !window.currentRoomInfo) return;

    const roomName = window.currentRoomInfo.name;
    const isOwner = window.currentRoomInfo.owner === window.currentUser.uid;
    const roomType = window.currentRoomInfo.type;
    let title = '';
    let instructions = '';

    if (isOwner) {
        // رسالة لمالك الغرفة
        if (roomType === 'teacher') {
            title = `أهلاً بك أيها الأستاذ في غرفة الدرس "${roomName}"`;
            instructions = `أنت تملك صلاحيات كاملة. استخدم لوحة التحكم في الأعلى لبدء بث مباشر، مشاركة سبورة، أو عرض ملف للطلاب.`;
        } else {
            title = `أهلاً بك، أنت مدير غرفة "${roomName}"`;
            instructions = `بصفتك مالك الغرفة، يمكنك استخدام لوحة التحكم لمشاركة فيديو يوتيوب، سبورة، أو ملف مع المشاركين.`;
        }
    } else {
        // رسالة للمنضمين العاديين
        title = `أهلاً بك في غرفة "${roomName}"`;
        instructions = `انتظر حتى يقوم مدير الغرفة بمشاركة محتوى الدرس. يمكنك استخدام الدردشة في الأسفل للتفاعل مع زملائك.`;
    }

    welcomeContainer.innerHTML = `
        <h2 style="font-family: 'Amiri', serif; font-size: 2em; color: var(--primary-color); margin-bottom: 15px;">${title}</h2>
        <p style="font-size: 1.1em; max-width: 600px; line-height: 1.7;">${instructions}</p>
    `;
}

// --- متغيرات عامة للسبورة ---
let whiteboard, ctx, isDrawing = false;
let startX = 0, startY = 0;
let currentTool = 'pen';
let brushColor = '#df4b26';
let brushSize = 5;
let isEraser = false;
let remoteDrawing = false; // لمنع إرسال البيانات التي نستقبلها مرة أخرى

/**
 * دالة التهيئة الرئيسية للسبورة.
 * يتم استدعاؤها مرة واحدة فقط عند فتح أداة السبورة.
 */
function initializeWhiteboard() {
    const canvasWrapper = document.getElementById('whiteboard-canvas-wrapper');
    whiteboard = document.getElementById('whiteboardCanvas');
    if (!whiteboard || !canvasWrapper) return;
    
    ctx = whiteboard.getContext('2d');

    // --- ضبط أبعاد الـ canvas ---
    function resizeCanvas() {
        whiteboard.width = canvasWrapper.offsetWidth;
        whiteboard.height = canvasWrapper.offsetHeight;
        redrawAllDrawings(); // إعادة رسم كل شيء عند تغيير الحجم
    }
    // استخدام ResizeObserver لمراقبة تغيير حجم الحاوية بشكل دقيق
    new ResizeObserver(resizeCanvas).observe(canvasWrapper);
    resizeCanvas(); // استدعاء فوري لضبط الحجم الأولي

    // --- ربط أحداث الفأرة واللمس ---
    whiteboard.addEventListener('mousedown', startDrawing);
    whiteboard.addEventListener('mousemove', draw);
    whiteboard.addEventListener('mouseup', stopDrawing);
    whiteboard.addEventListener('mouseout', stopDrawing);

    // --- تهيئة أزرار التحكم ---
    setupWhiteboardControls();

    // --- تهيئة مستمعات Firebase ---
    setupFirebaseWhiteboardListeners();
}

/**
 * ربط الأحداث لأزرار التحكم بالسبورة.
 */
function setupWhiteboardControls() {
    // اختيار الأداة (قلم، خط، مربع...)
    document.querySelectorAll('.wb-tool-btn').forEach(btn => {
        btn.addEventListener('click', (e) => {
            document.querySelectorAll('.wb-tool-btn').forEach(b => b.classList.remove('active'));
            e.currentTarget.classList.add('active');
            currentTool = e.currentTarget.dataset.tool;
            isEraser = false;
        });
    });

    // تغيير اللون والحجم
    document.getElementById('wbColorPicker').addEventListener('change', (e) => {
        isEraser = false;
        brushColor = e.target.value;
    });
    document.getElementById('wbBrushSize').addEventListener('change', (e) => brushSize = e.target.value);
    
    // تفعيل الممحاة
    document.getElementById('wbEraserBtn').addEventListener('click', () => {
        isEraser = true;
        currentTool = 'pen'; // الممحاة هي نفسها القلم
        document.querySelectorAll('.wb-tool-btn').forEach(b => b.classList.remove('active'));
        document.querySelector('.wb-tool-btn[data-tool="pen"]').classList.add('active');
    });
    
    // مسح السبورة بالكامل
    document.getElementById('wbClearBtn').addEventListener('click', () => {
        if (hasPermission('CONTROL_VIDEO')) {
            const whiteboardRef = window.ref(window.database, `rooms/${window.currentRoom}/whiteboard`);
            window.remove(whiteboardRef);
        } else {
            showGeneralNotification('error', 'فقط مدير الغرفة يمكنه مسح السبورة.');
        }
    });
}

/**
 * إعداد المستمعات التي تتلقى البيانات من Firebase.
 */
function setupFirebaseWhiteboardListeners() {
    const whiteboardRef = window.ref(window.database, `rooms/${window.currentRoom}/whiteboard`);
    
    const drawingListener = window.onValue(whiteboardRef, (snapshot) => {
        ctx.clearRect(0, 0, whiteboard.width, whiteboard.height); // مسح السبورة قبل إعادة الرسم
        const allDrawings = snapshot.val();
        if (allDrawings) {
            remoteDrawing = true;
            // الرسم بترتيب المفاتيح لضمان نفس الترتيب للجميع
            Object.keys(allDrawings).sort().forEach(key => {
                drawOnCanvas(allDrawings[key]);
            });
            remoteDrawing = false;
        }
    });
    window.videoSyncListeners.push({ ref: whiteboardRef, listener: drawingListener });
}

/**
 * يعيد رسم كل الأشكال المخزنة في Firebase. مفيد عند تغيير حجم النافذة.
 */
async function redrawAllDrawings() {
    if (!window.currentRoom || !ctx) return;
    const whiteboardRef = window.ref(window.database, `rooms/${window.currentRoom}/whiteboard`);
    const snapshot = await window.get(whiteboardRef);
    const allDrawings = snapshot.val();
    if (allDrawings) {
        Object.keys(allDrawings).sort().forEach(key => {
            drawOnCanvas(allDrawings[key]);
        });
    }
}

/**
 * تبدأ عملية الرسم عند الضغط على الفأرة.
 */
function startDrawing(e) {
    if (!hasPermission('CONTROL_VIDEO')) return;
    isDrawing = true;
    [startX, startY] = [e.offsetX, e.offsetY];

    if (currentTool === 'text') {
        const text = prompt("أدخل النص:", "");
        if (text) {
            const textData = {
                tool: 'text',
                text: text,
                x: startX,
                y: startY,
                color: brushColor,
                size: parseInt(brushSize) + 12
            };
            sendDrawingData(textData);
        }
        isDrawing = false;
    }
}

/**
 * تستمر في الرسم أثناء تحريك الفأرة (خاص بالقلم).
 */
function draw(e) {
    if (!isDrawing || !hasPermission('CONTROL_VIDEO')) return;

    if (currentTool === 'pen') {
        const penData = {
            tool: 'pen',
            x0: startX,
            y0: startY,
            x1: e.offsetX,
            y1: e.offsetY,
            color: isEraser ? '#FFFFFF' : brushColor,
            size: isEraser ? 30 : brushSize
        };
        sendDrawingData(penData);
        [startX, startY] = [e.offsetX, e.offsetY];
    }
}

/**
 * تنهي عملية الرسم عند رفع زر الفأرة (خاص بالأشكال).
 */
function stopDrawing(e) {
    if (!isDrawing || !hasPermission('CONTROL_VIDEO')) return;
    isDrawing = false;

    let shapeData = null;
    const endX = e.offsetX;
    const endY = e.offsetY;

    switch (currentTool) {
        case 'line':
            shapeData = { tool: 'line', x0: startX, y0: startY, x1: endX, y1: endY, color: brushColor, size: brushSize };
            break;
        case 'rectangle':
            shapeData = { tool: 'rectangle', x: startX, y: startY, width: endX - startX, height: endY - startY, color: brushColor, size: brushSize };
            break;
        case 'circle':
            const radius = Math.sqrt(Math.pow(endX - startX, 2) + Math.pow(endY - startY, 2));
            shapeData = { tool: 'circle', x: startX, y: startY, radius: radius, color: brushColor, size: brushSize };
            break;
    }

    if (shapeData) {
        sendDrawingData(shapeData);
    }
}

/**
 * دالة مركزية لإرسال بيانات الرسم إلى Firebase.
 */
function sendDrawingData(data) {
    if (remoteDrawing || !window.currentRoom) return;
    const whiteboardRef = window.ref(window.database, `rooms/${window.currentRoom}/whiteboard`);
    const newDrawingRef = window.push(whiteboardRef);
    window.set(newDrawingRef, data);
}

/**
 * دالة مركزية للرسم الفعلي على الـ canvas بناءً على البيانات.
 */
function drawOnCanvas(data) {
    if (!ctx || !data) return;
    ctx.strokeStyle = data.color;
    ctx.fillStyle = data.color;
    ctx.lineWidth = data.size;
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';

    switch (data.tool) {
        case 'pen':
        case 'line':
            ctx.beginPath();
            ctx.moveTo(data.x0, data.y0);
            ctx.lineTo(data.x1, data.y1);
            ctx.stroke();
            break;
        case 'rectangle':
            ctx.beginPath();
            ctx.strokeRect(data.x, data.y, data.width, data.height);
            break;
        case 'circle':
            ctx.beginPath();
            ctx.arc(data.x, data.y, data.radius, 0, 2 * Math.PI);
            ctx.stroke();
            break;
        case 'text':
            ctx.font = `${data.size}px Tajawal`;
            ctx.fillText(data.text, data.x, data.y);
            break;
    }
}

/**
 * رفع الملف المحدد إلى Google Drive.
 */
function uploadFileToDrive(file) {
    const metadata = {
        'name': `[StudyLink] ${file.name} (${window.currentRoom})`,
        'mimeType': file.type,
    };

    const form = new FormData();
    form.append('metadata', new Blob([JSON.stringify(metadata)], { type: 'application/json' }));
    form.append('file', file);

    const xhr = new XMLHttpRequest();
    xhr.open('POST', 'https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart' );
    xhr.setRequestHeader('Authorization', 'Bearer ' + gapi.client.getToken().access_token);
    
    const progressSection = document.getElementById('google-upload-progress');
    const progressBar = document.getElementById('upload-progress-bar');
    const percentageText = document.getElementById('upload-percentage');
    progressSection.classList.remove('hidden');

    xhr.upload.onprogress = (e) => {
        if (e.lengthComputable) {
            const percentage = Math.round((e.loaded / e.total) * 100);
            progressBar.style.width = percentage + '%';
            percentageText.textContent = percentage + '%';
        }
    };

    xhr.onload = () => {
        progressSection.classList.add('hidden');
        if (xhr.status === 200) {
            const response = JSON.parse(xhr.responseText);
            showGeneralNotification('success', 'تم رفع الملف بنجاح!');
            makeFilePublicAndShare(response.id);
        } else {
            showGeneralNotification('error', 'فشل رفع الملف. حاول مرة أخرى.');
            console.error('Upload failed:', xhr.responseText);
        }
    };
    
    xhr.send(form);
}
/**
 * ✅ النسخة النهائية والمُصححة:
 * تجعل الملف عاماً وتستخدم رابط التضمين (embed) الصحيح لعرضه داخل iframe.
 */
async function makeFilePublicAndShare(fileId) {
    // التأكد من أن Google API جاهز
    if (!gapiInited) {
        await initializeGapiClient();
    }
    // إذا فشلت التهيئة، توقف
    if (!gapiInited) {
        showGeneralNotification('error', 'لا يمكن مشاركة الملف. فشلت تهيئة API.');
        return;
    }

    try {
        // الخطوة 1: جعل الملف قابلاً للقراءة من قبل أي شخص (هذه الخطوة ضرورية)
        await gapi.client.drive.permissions.create({
            fileId: fileId,
            resource: { 'role': 'reader', 'type': 'anyone' }
        });

        // الخطوة 2: ✅ بناء رابط التضمين الصحيح والمباشر.
        // هذا هو الرابط الذي صممته Google ليتم وضعه داخل <iframe>.
        const embedUrl = `https://drive.google.com/file/d/${fileId}/preview`;

        // الخطوة 3: الحصول على نوع الملف (mimeType ) من أجل معلومات إضافية إذا أردنا
        const fileMetadata = await gapi.client.drive.files.get({
            fileId: fileId,
            fields: 'mimeType'
        });
        const mimeType = fileMetadata.result.mimeType;

        // الخطوة 4: مشاركة الرابط الصحيح في قاعدة بيانات Firebase
        const sharedFileRef = window.ref(window.database, `rooms/${window.currentRoom}/sharedFile`);
        window.set(sharedFileRef, {
            url: embedUrl, // <-- استخدام رابط التضمين الصحيح
            mimeType: mimeType,
            uploader: window.currentUser.displayName
        });

        showGeneralNotification('success', 'تمت مشاركة الملف بنجاح مع الغرفة.');

    } catch (error) {
        console.error("Sharing/Permission Error:", error);
        showGeneralNotification('error', 'فشل في مشاركة الملف. تأكد من صلاحيات حسابك.');
    }
}

async function startStream() {
    try {
        const [screenShare, micShare, camShare] = await Promise.all([
            navigator.mediaDevices.getDisplayMedia({ video: true, audio: true }),
            navigator.mediaDevices.getUserMedia({ video: false, audio: true }),
            navigator.mediaDevices.getUserMedia({ video: true, audio: false })
        ]);
        localScreenStream = screenShare;
        localCamStream = camShare;
        const audioContext = new AudioContext();
        const destination = audioContext.createMediaStreamDestination();
        if (micShare.getAudioTracks().length > 0) audioContext.createMediaStreamSource(micShare).connect(destination);
        if (screenShare.getAudioTracks().length > 0) audioContext.createMediaStreamSource(screenShare).connect(destination);
        const finalAudioTrack = destination.stream.getAudioTracks()[0];
        localScreenStream.getAudioTracks().forEach(track => localScreenStream.removeTrack(track));
        if (finalAudioTrack) localScreenStream.addTrack(finalAudioTrack);
        localCamStream.getVideoTracks()[0].enabled = false;
        document.getElementById('main-stream-video').srcObject = localScreenStream;
        document.getElementById('pip-stream-video').srcObject = localCamStream;
        document.getElementById('teacher-pip-message').classList.remove('hidden');
        document.getElementById('pip-stream-video').style.display = 'none';
        
        updateStreamUI(true);

        // *** التعديل هنا: نضبط حالة البث إلى "نشط" ***
        const streamStateRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/isActive`);
        await window.set(streamStateRef, true);

        const camStateRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/state/isCamActive`);
        window.set(camStateRef, false);
        
        const users = window.currentRoomInfo.users || {};
        for (const userId in users) {
            if (userId !== window.currentUser.uid) createPeerConnection(userId, true);
        }
    } catch (error) {
        console.error("فشل بدء البث:", error);
        showGeneralNotification('error', 'فشل بدء البث. تأكد من منح جميع الأذونات.');
        stopStream(); // استدعاء دالة الإيقاف الجديدة لتنظيف الحالة
    }
}

function stopStream() {
    // إيقاف المسارات المحلية (الكاميرا والشاشة)
    if (localScreenStream) {
        localScreenStream.getTracks().forEach(track => track.stop());
        localScreenStream = null;
    }
    if (localCamStream) {
        localCamStream.getTracks().forEach(track => track.stop());
        localCamStream = null;
    }

    // إغلاق جميع اتصالات WebRTC الحالية
    Object.values(peerConnections).forEach(pc => pc.close());
    peerConnections = {};

    // تحديث واجهة المستخدم لتعكس حالة إيقاف البث
    updateStreamUI(false);

    // *** التعديل الجوهري هنا ***
    // بدلاً من حذف المسار بالكامل، نقوم بتحديث حالته وننظف البيانات القديمة
    if (window.currentRoom) {
        const updates = {};
        // نضبط حالة البث على "غير نشط"
        updates[`/rooms/${window.currentRoom}/liveStream/isActive`] = false;
        // نحذف العروض ومرشحي ICE القدامى لضمان عدم حدوث تضارب عند إعادة البث
        updates[`/rooms/${window.currentRoom}/liveStream/offers`] = null;
        updates[`/rooms/${window.currentRoom}/liveStream/answers`] = null;
        updates[`/rooms/${window.currentRoom}/liveStream/iceCandidates`] = null;
        
        // تنفيذ التحديثات دفعة واحدة
        window.update(window.ref(window.database), updates)
            .then(() => {
                console.log("Live stream state updated to inactive.");
            })
            .catch(error => {
                console.error("Failed to update stream state:", error);
            });
    }
}




const toggleTrack = (kind) => () => {
    const btn = document.getElementById(kind === 'audio' ? 'toggleMicBtn' : 'toggleCamBtn');
    if (!btn) return;
    let track;
    if (kind === 'audio' && localScreenStream) track = localScreenStream.getAudioTracks()[0];
    else if (kind === 'video' && localCamStream) track = localCamStream.getVideoTracks()[0];
    if (!track) return;
    track.enabled = !track.enabled;
    const icon = btn.querySelector('i');
    btn.style.background = track.enabled ? 'var(--success-color)' : 'var(--danger-color)';
    icon.className = `fa fa-${kind === 'audio' ? 'microphone' : 'microphone-slash'}`;
    if (kind === 'video') {
        icon.className = `fa fa-${track.enabled ? 'video' : 'video-slash'}`;
        document.getElementById('pip-stream-video').style.display = track.enabled ? 'block' : 'none';
        document.getElementById('teacher-pip-message').classList.toggle('hidden', track.enabled);
        if (window.currentRoom) {
            const camStateRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/state/isCamActive`);
            window.set(camStateRef, track.enabled);
        }
    }
};

async function createPeerConnection(peerId, isInitiator) {
    if (peerConnections[peerId]) return;
    const pc = new RTCPeerConnection(stunServers);
    peerConnections[peerId] = pc;
    if (isInitiator) {
        [localScreenStream, localCamStream].forEach(s => s?.getTracks().forEach(t => pc.addTrack(t, s)));
    } else {
        const mainVideo = document.getElementById('main-stream-video');
        const unmuteBtn = document.getElementById('unmute-btn');
        unmuteBtn.onclick = () => { mainVideo.muted = false; unmuteBtn.classList.add('hidden'); };
        pc.ontrack = (event) => {
            const stream = event.streams[0];
            if (stream.getVideoTracks()[0]?.contentHint === 'camera' || stream.getAudioTracks().length === 0) {
                document.getElementById('pip-stream-video').srcObject = stream;
            } else {
                mainVideo.srcObject = stream;
                unmuteBtn.classList.remove('hidden');
            }
            document.getElementById('live-stream-placeholder').classList.add('hidden');
        };
    }
    pc.onicecandidate = e => {
        if (e.candidate) {
            const iceRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/iceCandidates/${peerId}`);
            window.push(iceRef, { candidate: e.candidate.toJSON(), peerId: window.currentUser.uid });
        }
    };
    if (isInitiator) {
        const offer = await pc.createOffer();
        await pc.setLocalDescription(offer);
        const offerRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/offers/${peerId}`);
        window.set(offerRef, { sdp: offer.sdp, type: offer.type });
    }
}
      function setupFirebaseSignaling() {
    const roomRef = window.ref(window.database, `rooms/${window.currentRoom}`);
    const ownerId = window.currentRoomInfo.owner;
    const myId = window.currentUser.uid;

    if (!hasPermission('CONTROL_VIDEO')) {
        const camStateRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/state/isCamActive`);
        const camStateListener = window.onValue(camStateRef, (snapshot) => {
            const isCamActive = snapshot.val();
            const pipVideo = document.getElementById('pip-stream-video');
            const studentPlaceholder = document.getElementById('student-pip-placeholder');
            if (pipVideo && studentPlaceholder) {
                pipVideo.style.display = isCamActive ? 'block' : 'none';
                studentPlaceholder.classList.toggle('hidden', isCamActive);
            }
        });
        window.videoSyncListeners.push({ ref: camStateRef, listener: camStateListener });
    }

    if (hasPermission('CONTROL_VIDEO')) {
        const usersRef = window.child(roomRef, 'users');
        const usersListener = window.onValue(usersRef, (snapshot) => {
            const users = snapshot.val();
            if (users && (localScreenStream || localCamStream)) {
                for (const userId in users) {
                    if (userId !== myId && !peerConnections[userId]) createPeerConnection(userId, true);
                }
            }
        });
        window.videoSyncListeners.push({ ref: usersRef, listener: usersListener });
    }

    // *** التعديل الجوهري هنا: منطق الطالب يعتمد على حالة `isActive` ***
    if (!hasPermission('CONTROL_VIDEO')) {
        const streamStateRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/isActive`);
        const streamStateListener = window.onValue(streamStateRef, (snapshot) => {
            const isStreaming = snapshot.val(); // ستكون القيمة true أو false أو null
            const ownerId = window.currentRoomInfo.owner;

            if (isStreaming === true) {
                // إذا كان البث نشطًا ولم نكن متصلين بعد، نبدأ الاتصال
                if (!peerConnections[ownerId]) {
                    console.log("Stream is active. Creating peer connection.");
                    createPeerConnection(ownerId, false);
                }
            } else {
                // إذا كان البث غير نشط (false أو null)، نغلق الاتصال ونعيد الواجهة
                console.log("Stream is inactive. Closing peer connection.");
                if (peerConnections[ownerId]) {
                    peerConnections[ownerId].close();
                    delete peerConnections[ownerId];
                }
                updateStreamUI(false);
            }
        });
        window.videoSyncListeners.push({ ref: streamStateRef, listener: streamStateListener });
    }

    const offerRef = window.child(roomRef, `liveStream/offers/${myId}`);
    const offerListener = window.onValue(offerRef, async (snapshot) => {
        if (snapshot.exists() && !hasPermission('CONTROL_VIDEO')) {
            await handleOffer(snapshot.val(), ownerId);
            window.remove(snapshot.ref);
        }
    });
    window.videoSyncListeners.push({ ref: offerRef, listener: offerListener });

    const answerRef = window.child(roomRef, `liveStream/answers/${myId}`);
    const answerListener = window.onValue(answerRef, async (snapshot) => {
        if (snapshot.exists() && hasPermission('CONTROL_VIDEO')) {
            const answers = snapshot.val();
            for (const studentId in answers) { 
                await handleAnswer(answers[studentId], studentId);
            }
            window.remove(snapshot.ref);
        }
    });
    window.videoSyncListeners.push({ ref: answerRef, listener: answerListener });

    const iceRef = window.child(roomRef, `liveStream/iceCandidates/${myId}`);
    const iceListener = window.onValue(iceRef, async (snapshot) => {
        if (snapshot.exists()) {
            const candidates = snapshot.val();
            for (const key in candidates) {
                const data = candidates[key];
                const targetPeerId = hasPermission('CONTROL_VIDEO') ? data.peerId : ownerId;
                await handleAddIceCandidate(data.candidate, targetPeerId);
            }
            window.remove(snapshot.ref);
        }
    });
    window.videoSyncListeners.push({ ref: iceRef, listener: iceListener });
}



async function handleOffer(offer, peerId) {
    if (!peerConnections[peerId]) {
        createPeerConnection(peerId, false);
    }
    const pc = peerConnections[peerId];
    await pc.setRemoteDescription(new RTCSessionDescription(offer));
    const answer = await pc.createAnswer();
    await pc.setLocalDescription(answer);
    
    // ✅ الإصلاح: الكتابة في المسار الصحيح الذي تسمح به القواعد
    const answerRef = window.ref(window.database, `rooms/${window.currentRoom}/liveStream/answers/${peerId}/${window.currentUser.uid}`);
    window.set(answerRef, { sdp: answer.sdp, type: answer.type });
}

async function handleAnswer(answer, peerId) {
    const pc = peerConnections[peerId];
    if (pc && !pc.currentRemoteDescription) {
        await pc.setRemoteDescription(new RTCSessionDescription(answer));
    }
}

async function handleAddIceCandidate(candidate, peerId) {
    const pc = peerConnections[peerId];
    if (pc && pc.remoteDescription) {
        try { await pc.addIceCandidate(new RTCIceCandidate(candidate)); }
        catch (e) { console.error('Error adding ICE candidate', e); }
    }
}

function updateStreamUI(isStreaming) {
    document.getElementById('start-stream-options').classList.toggle('hidden', isStreaming);
    document.getElementById('in-stream-controls').classList.toggle('hidden', !isStreaming);
    document.getElementById('live-stream-placeholder').classList.toggle('hidden', isStreaming);
    if (!isStreaming) {
        document.getElementById('main-stream-video').srcObject = null;
        document.getElementById('pip-stream-video').srcObject = null;
        document.getElementById('teacher-pip-message').classList.add('hidden');
        document.getElementById('student-pip-placeholder').classList.add('hidden');
        document.getElementById('pip-stream-video').style.display = 'none';
        document.getElementById('unmute-btn').classList.add('hidden');
    }
}
